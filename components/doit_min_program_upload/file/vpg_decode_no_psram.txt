/* 你项目里已有的 JPEG 解码函数，保持签名不变 */
static jpeg_error_t decode_jpg_from_mem(const uint8_t *jpg_data, const int jpg_len, lv_image_dsc_t *output_dsc)
{

    // ESP_LOGI(TAG, "decode open： %ld", (uint32_t)esp_timer_get_time());
    jpeg_error_t ret = JPEG_ERR_OK;

    // 分配到栈里，避免堆分配
    jpeg_dec_io_t jpeg_io = {0};
    jpeg_dec_header_info_t out_info = {0};

    memset(&jpeg_io, 0, sizeof(jpeg_dec_io_t));
    memset(&out_info, 0, sizeof(jpeg_dec_header_info_t));

    jpeg_io.inbuf = jpg_data;
    jpeg_io.inbuf_len = jpg_len;

    ret = jpeg_dec_parse_header(jpeg_dec, &jpeg_io, &out_info);
    if (ret != JPEG_ERR_OK)
        return ret;

    output_dsc->header.w = out_info.width;
    output_dsc->header.h = out_info.height;
    if (config.output_type == JPEG_PIXEL_FORMAT_RGB565_LE ||
        config.output_type == JPEG_PIXEL_FORMAT_RGB565_BE ||
        config.output_type == JPEG_PIXEL_FORMAT_CbYCrY)
    {
        output_dsc->data_size = out_info.width * out_info.height * 2;
    }
    else if (config.output_type == JPEG_PIXEL_FORMAT_RGB888)
    {
        output_dsc->data_size = out_info.width * out_info.height * 3;
    }
    else
    {
        ret = JPEG_ERR_INVALID_PARAM;
        return ret;
    }

    output_dsc->header.cf = LV_COLOR_FORMAT_RGB565; // 必须设

    jpeg_io.outbuf = output_dsc->data;
    ret = jpeg_dec_process(jpeg_dec, &jpeg_io);

    // ESP_LOGI(TAG, "img decode success %dx%d  %lu bytes", out_info.width, out_info.height, output_dsc->data_size);

    // ESP_LOGI(TAG, "decode over： %ld", (uint32_t)esp_timer_get_time());
    return ret;
}

static const char *TAG = "VPG_FILE";

/* =================== 可调参数 =================== */
#define VPG_BUFF_NUM 2             /* 显示用双缓冲 */
#define VPG_QUEUE_LEN 1            /* 只保留最新一帧 */
#define VPG_SHOW_TASK_STACK 1280   /* words */
#define VPG_DECODE_TASK_STACK 2048 /* words */
#define VPG_TASK_PRIO 6
#define VPG_TASK_CORE 1 /* 绑定核；如无 SMP 可忽略 */
#define VPG_MIN_FPS 5
#define VPG_MAX_FPS 60
/* =============================================== */

/* 播放控制消息（解码->显示） */
typedef struct
{
    uint8_t buf_idx;
} vpg_frame_t;

/* 播放器主状态 */
typedef struct lv_vpg_t
{
    char spiff_path[64];
    bool is_vpg_player;

    /* LVGL 相关 */
    lv_obj_t *img_obj;
    lv_image_dsc_t lv_buf[VPG_BUFF_NUM]; /* 双显示缓冲（RGB565） */
    uint8_t lv_buf_cur_idx;              /* 下一帧解码写入的缓冲索引 */

    /* 文件与索引（PSRAM） */
    FILE *fp;
    FileHeader fileHeader;
    ItemHeader *ItemHeader; /* 索引表复制到 PSRAM */
    uint16_t cur_frame_idx;

    /* 压缩帧 staging（PSRAM，复用） */
    uint8_t *comp_buf;
    uint32_t comp_buf_size;
} lv_vpg_t;

/* ==== 全局/静态资源 ==== */
static lv_vpg_t *vpg = NULL;
static bool vpg_is_init = false;

static TaskHandle_t s_vpg_decode_task = NULL;
static TaskHandle_t s_vpg_show_task = NULL;

static QueueHandle_t s_vpg_queue = NULL;                    /* vpg_frame_t 队列：解码->显示（长度1，覆盖） */
static SemaphoreHandle_t s_buf_free[VPG_BUFF_NUM] = {NULL}; /* 双缓冲可用信号量 */

/* ===================== 工具函数 ===================== */
static inline void queue_overwrite_latest(QueueHandle_t q, const void *item)
{
    /* 长度1队列的“覆盖式写入”：尝试一次，失败则先receive丢旧值再发送 */
    if (xQueueSend(q, item, 0) != pdTRUE)
    {
        vpg_frame_t dummy;
        xQueueReceive(q, &dummy, 0); /* 丢弃旧的 */
        xQueueSend(q, item, 0);
    }
}

/* ===================== 显示任务 ===================== */
static void vpg_show_task(void *arg)
{
    vpg_frame_t msg;
    int last_idx = -1;

    for (;;)
    {
        if (!vpg || !vpg->is_vpg_player)
        {
            vTaskDelay(pdMS_TO_TICKS(10));
            continue;
        }

        if (xQueueReceive(s_vpg_queue, &msg, pdMS_TO_TICKS(10)) == pdTRUE)
        {
            uint8_t idx = msg.buf_idx;

            lvgl_port_lock(-1);
            lv_image_set_src(vpg->img_obj, &vpg->lv_buf[idx]);
            lv_obj_invalidate(vpg->img_obj);
            lvgl_port_unlock();

            /* 释放上一帧缓冲 */
            if (last_idx >= 0)
                xSemaphoreGive(s_buf_free[last_idx]);
            last_idx = idx;
        }
    }
}

/* ===================== 解码任务 ===================== */
static void vpg_decode_task(void *arg)
{
    for (;;)
    {
        if (!vpg || !vpg->is_vpg_player)
        {
            vTaskDelay(pdMS_TO_TICKS(10));
            continue;
        }

        /* 当前要写入的显示缓冲 */
        uint8_t idx = vpg->lv_buf_cur_idx;

        /* 等显示任务释放这块缓冲 */
        xSemaphoreTake(s_buf_free[idx], portMAX_DELAY);

        /* 定位到当前帧 */
        uint32_t off = vpg->ItemHeader[vpg->cur_frame_idx].offset;
        uint32_t size = vpg->ItemHeader[vpg->cur_frame_idx].size;

        if (size == 0 || size > vpg->comp_buf_size)
        {
            ESP_LOGE(TAG, "Invalid frame size=%u (max=%u) at #%u", (unsigned)size, (unsigned)vpg->comp_buf_size, vpg->cur_frame_idx);
            xSemaphoreGive(s_buf_free[idx]);
            /* 跳过该帧 */
            vpg->cur_frame_idx = (vpg->cur_frame_idx + 1) % vpg->fileHeader.itemNum;
            continue;
        }

        if (fseek(vpg->fp, (long)off, SEEK_SET) != 0)
        {
            ESP_LOGE(TAG, "fseek offset=%u failed, errno=%d", (unsigned)off, errno);
            xSemaphoreGive(s_buf_free[idx]);
            vpg->cur_frame_idx = (vpg->cur_frame_idx + 1) % vpg->fileHeader.itemNum;
            continue;
        }

        size_t rd = fread(vpg->comp_buf, 1, size, vpg->fp);
        if (rd != size)
        {
            ESP_LOGE(TAG, "fread want=%u got=%u failed, errno=%d", (unsigned)size, (unsigned)rd, errno);
            xSemaphoreGive(s_buf_free[idx]);
            vpg->cur_frame_idx = (vpg->cur_frame_idx + 1) % vpg->fileHeader.itemNum;
            continue;
        }

        /* 准备输出指向显示缓冲（很关键） */
        vpg->lv_buf[idx].data = vpg->lv_buf[idx].data; /* 占位，强调 data 已指向预分配的显示缓冲 */
        /* decode_jpg_from_mem 需要 output_dsc->data 指向目标缓冲 */
        /* 若你的 decode_jpg_from_mem 内部没有重置 data，请确保这里赋值 */
        /* vpg->lv_buf[idx].data 已在 init 时分配，这里无需再 malloc */

        jpeg_error_t jrc = decode_jpg_from_mem(vpg->comp_buf, (int)size, &vpg->lv_buf[idx]);
        if (jrc == JPEG_ERR_OK)
        {
            /* 通知显示任务使用 idx */
            vpg_frame_t out = {.buf_idx = idx};
            queue_overwrite_latest(s_vpg_queue, &out);
        }
        else
        {
            ESP_LOGE(TAG, "JPEG decode failed at frame #%u", vpg->cur_frame_idx);
            xSemaphoreGive(s_buf_free[idx]); /* 归还缓冲 */
        }

        /* 下一帧/下一个缓冲 */
        vpg->lv_buf_cur_idx = (uint8_t)((idx + 1) % VPG_BUFF_NUM);
        vpg->cur_frame_idx = (uint16_t)((vpg->cur_frame_idx + 1) % vpg->fileHeader.itemNum);

        /* 帧率控制（容错夹紧） */
        uint32_t fps = vpg->fileHeader.fps;
        if (fps < VPG_MIN_FPS)
            fps = VPG_MIN_FPS;
        if (fps > VPG_MAX_FPS)
            fps = VPG_MAX_FPS;
        vTaskDelay(pdMS_TO_TICKS(1000 / fps));
    }
}

/* ===================== 对外 API ===================== */

void doit_vpg_player_stop(void)
{
    if (!vpg || !vpg->is_vpg_player)
        return;

    vpg->is_vpg_player = false;

    if (s_vpg_decode_task)
        vTaskSuspend(s_vpg_decode_task);
    if (s_vpg_show_task)
        vTaskSuspend(s_vpg_show_task);

    /* 释放所有显示缓冲令其可再次获取 */
    for (int i = 0; i < VPG_BUFF_NUM; ++i)
    {
        if (s_buf_free[i])
            xSemaphoreGive(s_buf_free[i]);
    }

    /* 关闭文件与临时缓冲（可选：也可保留，加速下次启动） */
    if (vpg->fp)
    {
        fclose(vpg->fp);
        vpg->fp = NULL;
    }
    if (vpg->comp_buf)
    {
        heap_caps_free(vpg->comp_buf);
        vpg->comp_buf = NULL;
    }
    if (vpg->ItemHeader)
    {
        heap_caps_free(vpg->ItemHeader);
        vpg->ItemHeader = NULL;
    }

    ESP_LOGI(TAG, "VPG player stopped. free8=%u largest8=%u psram=%u",
             (unsigned)heap_caps_get_free_size(MALLOC_CAP_8BIT),
             (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_8BIT),
             (unsigned)heap_caps_get_free_size(MALLOC_CAP_SPIRAM));
}

void doit_vpg_player_start(const char *dir_name)
{
    ESP_LOGI(TAG, "加载的动画文件：%s", dir_name);

    if (!vpg_is_init)
    {
        /* 首次整体初始化（只做一次） */
        vpg = (lv_vpg_t *)heap_caps_calloc(1, sizeof(lv_vpg_t), MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
        if (!vpg)
        {
            ESP_LOGE(TAG, "alloc vpg struct failed");
            return;
        }

        s_vpg_queue = xQueueCreate(VPG_QUEUE_LEN, sizeof(vpg_frame_t));
        if (!s_vpg_queue)
        {
            ESP_LOGE(TAG, "create vpg queue failed");
            return;
        }

        vpg->img_obj = doit_ui_get_show_lv_obj();

        /* 屏幕尺寸与双显示缓冲的分配（RGB565） */
        uint16_t width = 0, height = 0;
        doit_get_ui_screen_size(&width, &height);
        uint32_t pix_bytes = (uint32_t)width * (uint32_t)height * 2;

        for (int i = 0; i < VPG_BUFF_NUM; ++i)
        {
            vpg->lv_buf[i].data = (uint8_t *)heap_caps_aligned_alloc(
                16, pix_bytes, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
            if (!vpg->lv_buf[i].data)
            {
                ESP_LOGE(TAG, "alloc lv_buf[%d] failed (%u bytes)", i, (unsigned)pix_bytes);
                return;
            }
            vpg->lv_buf[i].data_size = pix_bytes; /* 初值：整屏大小 */
            vpg->lv_buf[i].header.w = width;
            vpg->lv_buf[i].header.h = height;
            vpg->lv_buf[i].header.cf = LV_COLOR_FORMAT_RGB565;

            s_buf_free[i] = xSemaphoreCreateBinary();
            xSemaphoreGive(s_buf_free[i]); /* 初始可用 */
        }

        /* 创建任务（先挂起，待文件/索引就绪再唤醒） */
        BaseType_t rc = xTaskCreatePinnedToCore(vpg_show_task, "vpg_show_task",
                                                VPG_SHOW_TASK_STACK, NULL, VPG_TASK_PRIO,
                                                &s_vpg_show_task, VPG_TASK_CORE);
        if (rc != pdPASS)
            ESP_LOGE(TAG, "create vpg_show_task failed");
        else
            vTaskSuspend(s_vpg_show_task);

        rc = xTaskCreatePinnedToCore(vpg_decode_task, "vpg_decode_task",
                                     VPG_DECODE_TASK_STACK, NULL, VPG_TASK_PRIO,
                                     &s_vpg_decode_task, VPG_TASK_CORE);
        if (rc != pdPASS)
            ESP_LOGE(TAG, "create vpg_decode_task failed");
        else
            vTaskSuspend(s_vpg_decode_task);

        vpg_is_init = true;
        ESP_LOGI(TAG, "VPG init done. free8=%u largest8=%u psram=%u",
                 (unsigned)heap_caps_get_free_size(MALLOC_CAP_8BIT),
                 (unsigned)heap_caps_get_largest_free_block(MALLOC_CAP_8BIT),
                 (unsigned)heap_caps_get_free_size(MALLOC_CAP_SPIRAM));
    }

    /* 运行期参数 */
    vpg->is_vpg_player = true;
    vpg->lv_buf_cur_idx = 0;
    vpg->cur_frame_idx = 0;
    snprintf(vpg->spiff_path, sizeof(vpg->spiff_path), "/littlefs/%s", dir_name);

    /* 打开文件，读取头与索引 */
    vpg->fp = fopen(vpg->spiff_path, "rb");
    if (!vpg->fp)
    {
        ESP_LOGE(TAG, "fopen %s failed: %s (errno=%d)", vpg->spiff_path, strerror(errno), errno);
        return;
    }

    if (fread(&vpg->fileHeader, sizeof(FileHeader), 1, vpg->fp) != 1)
    {
        ESP_LOGE(TAG, "read FileHeader failed");
        fclose(vpg->fp);
        vpg->fp = NULL;
        return;
    }
    if (vpg->fileHeader.magic != 0xAABBCCDD)
    {
        ESP_LOGE(TAG, "magic wrong: not VPG");
        fclose(vpg->fp);
        vpg->fp = NULL;
        return;
    }

    size_t idx_bytes = sizeof(ItemHeader) * vpg->fileHeader.itemNum;
    vpg->ItemHeader = (ItemHeader *)heap_caps_malloc(idx_bytes, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!vpg->ItemHeader)
    {
        ESP_LOGE(TAG, "alloc ItemHeader failed (%u bytes)", (unsigned)idx_bytes);
        fclose(vpg->fp);
        vpg->fp = NULL;
        return;
    }
    if (fread(vpg->ItemHeader, idx_bytes, 1, vpg->fp) != 1)
    {
        ESP_LOGE(TAG, "read ItemHeader failed");
        heap_caps_free(vpg->ItemHeader);
        vpg->ItemHeader = NULL;
        fclose(vpg->fp);
        vpg->fp = NULL;
        return;
    }

    /* 计算最大帧大小并分配复用 staging 缓冲 */
    uint32_t max_fr = 0;
    for (uint32_t i = 0; i < vpg->fileHeader.itemNum; ++i)
    {
        if (vpg->ItemHeader[i].size > max_fr)
            max_fr = vpg->ItemHeader[i].size;
    }
    vpg->comp_buf_size = max_fr;
    vpg->comp_buf = (uint8_t *)heap_caps_malloc(vpg->comp_buf_size, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!vpg->comp_buf)
    {
        ESP_LOGE(TAG, "alloc comp_buf failed (%u bytes)", (unsigned)max_fr);
        heap_caps_free(vpg->ItemHeader);
        vpg->ItemHeader = NULL;
        fclose(vpg->fp);
        vpg->fp = NULL;
        return;
    }

    ESP_LOGI(TAG, "VPG open ok: frames=%lu fps=%lu max_frame=%u",
             vpg->fileHeader.itemNum, vpg->fileHeader.fps, (unsigned)max_fr);

    /* 唤醒任务开始工作 */
    if (s_vpg_show_task)
        vTaskResume(s_vpg_show_task);
    if (s_vpg_decode_task)
        vTaskResume(s_vpg_decode_task);
}
